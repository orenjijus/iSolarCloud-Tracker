Parameter_ViewType = 
DATATABLE(
    "ViewType", STRING,
    "ViewTypeValue", INTEGER,
    {
        {"Actual", 0},
        {"Adjusted", 1}
    }
)

-- ============================================
-- STEP 2: HELPER MEASURE
-- ============================================
-- PowerBI: Modeling > New Measure
-- Name: _ShouldExcludeIssueDates

_ShouldExcludeIssueDates = 
VAR SelectedView = SELECTEDVALUE(Parameter_ViewType[ViewTypeValue], 0)
RETURN
    IF(SelectedView = 1, TRUE(), FALSE())

-- ============================================
-- STEP 3: ENERGY MEASURES
-- ============================================

-- Energy MWh
Energy MWh = 
VAR ExcludeIssues = [_ShouldExcludeIssueDates]
RETURN
    IF(
        ExcludeIssues,
        CALCULATE(
            SUM('mart mart_site_performance_daily'[daily_energy_mwh]),
            'mart mart_site_performance_daily'[is_issue_date] = FALSE()
        ),
        SUM('mart mart_site_performance_daily'[daily_energy_mwh])
    )

-- Energy Monthly MWh
Energy Monthly MWh = 
VAR ExcludeIssues = [_ShouldExcludeIssueDates]
VAR CurrentYear = SELECTEDVALUE('mart mart_site_performance_daily'[year])
VAR CurrentMonth = SELECTEDVALUE('mart mart_site_performance_daily'[month])
RETURN
    IF(
        ExcludeIssues,
        CALCULATE(
            SUM('mart mart_site_performance_daily'[daily_energy_mwh]),
            'mart mart_site_performance_daily'[is_issue_date] = FALSE(),
            'mart mart_site_performance_daily'[year] = CurrentYear,
            'mart mart_site_performance_daily'[month] = CurrentMonth
        ),
        CALCULATE(
            SUM('mart mart_site_performance_daily'[daily_energy_mwh]),
            'mart mart_site_performance_daily'[year] = CurrentYear,
            'mart mart_site_performance_daily'[month] = CurrentMonth
        )
    )

-- Energy YTD MWh
Energy YTD MWh = 
VAR ExcludeIssues = [_ShouldExcludeIssueDates]
VAR CurrentYear = SELECTEDVALUE('mart mart_site_performance_daily'[year])
VAR MaxDate = MAX('mart mart_site_performance_daily'[date_key])
RETURN
    IF(
        ExcludeIssues,
        CALCULATE(
            SUM('mart mart_site_performance_daily'[daily_energy_mwh]),
            'mart mart_site_performance_daily'[is_issue_date] = FALSE(),
            'mart mart_site_performance_daily'[year] = CurrentYear,
            'mart mart_site_performance_daily'[date_key] <= MaxDate
        ),
        CALCULATE(
            SUM('mart mart_site_performance_daily'[daily_energy_mwh]),
            'mart mart_site_performance_daily'[year] = CurrentYear,
            'mart mart_site_performance_daily'[date_key] <= MaxDate
        )
    )

-- STEP 9: COUNT MEASURES
-- ============================================

-- Total Days
Total Days = 
VAR ExcludeIssues = [_ShouldExcludeIssueDates]
RETURN
    IF(
        ExcludeIssues,
        CALCULATE(
            COUNTROWS('mart mart_site_performance_daily'),
            'mart mart_site_performance_daily'[is_issue_date] = FALSE()
        ),
        COUNTROWS('mart mart_site_performance_daily')
    )

-- Issue Days Count
Issue Days Count = 
COUNTROWS(
    FILTER(
        'mart mart_site_performance_daily',
        'mart mart_site_performance_daily'[is_issue_date] = TRUE()
    )
)

-- Adjusted Days Count
Adjusted Days Count = 
COUNTROWS(
    FILTER(
        'mart mart_site_performance_daily',
        'mart mart_site_performance_daily'[is_issue_date] = FALSE()
    )
)

-- Issue Days %
Issue Days % = 
DIVIDE(
    [Issue Days Count],
    COUNTROWS('mart mart_site_performance_daily'),
    0
) * 100
